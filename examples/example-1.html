Пример 1: Банковские счета

ООП + коллекции + Stream API в одном файле

Консольное приложение учёта банковских счетов: создание счетов, пополнение, перевод, выписка и отчёты. Демонстрирует классы, record, HashMap, пользовательские исключения, текстовые блоки, switch-меню и Stream API.

Полный код (BankApp.java)

java
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.*;
import java.util.stream.Collectors;

public class BankApp {

    // ---------- Доменная модель ----------

    /** Счёт: неизменяемый идентификатор + изменяемый баланс под блокировкой. */
    static final class Account {
        private static long counter = 1000;
        final long id;
        final String owner;
        private BigDecimal balance = BigDecimal.ZERO;

        Account(String owner) {
            this.id = ++counter;
            this.owner = owner;
        }

        synchronized void deposit(BigDecimal amount) {
            if (amount.signum() <= 0)
                throw new IllegalArgumentException("Сумма должна быть положительной");
            balance = balance.add(amount);
        }

        synchronized void withdraw(BigDecimal amount) {
            if (amount.signum() <= 0)
                throw new IllegalArgumentException("Сумма должна быть положительной");
            if (balance.compareTo(amount) < 0)
                throw new InsufficientFundsException(this, amount);
            balance = balance.subtract(amount);
        }

        synchronized BigDecimal getBalance() { return balance; }

        @Override public String toString() {
            return "Счёт #%d (%s): %s ₽".formatted(id, owner,
                    balance.setScale(2, RoundingMode.HALF_UP));
        }
    }

    /** Проверенное исключение предметной области. */
    static class InsufficientFundsException extends RuntimeException {
        InsufficientFundsException(Account acc, BigDecimal amount) {
            super("Недостаточно средств на счёте #%d: нужно %s, доступно %s"
                    .formatted(acc.id, amount, acc.getBalance()));
        }
    }

    /** Транзакция — record: неизменяемый «переносчик данных». */
    record Transaction(long fromId, long toId, BigDecimal amount, String timestamp) {}

    // ---------- «База данных» ----------

    private final Map<Long, Account> accounts = new LinkedHashMap<>();
    private final List<Transaction> history = new ArrayList<>();

    Account openAccount(String owner) {
        Account acc = new Account(owner);
        accounts.put(acc.id, acc);
        System.out.println("Открыт " + acc);
        return acc;
    }

    void transfer(long fromId, long toId, BigDecimal amount) {
        Account from = require(fromId);
        Account to = require(toId);
        from.withdraw(amount);
        try {
            to.deposit(amount);
        } catch (RuntimeException e) {
            from.deposit(amount);   // откат при ошибке зачисления
            throw e;
        }
        history.add(new Transaction(fromId, toId, amount,
                java.time.LocalDateTime.now().toString()));
        System.out.printf("Переведено %s ₽: #%d → #%d%n", amount, fromId, toId);
    }

    private Account require(long id) {
        Account acc = accounts.get(id);
        if (acc == null) throw new NoSuchElementException("Счёт #" + id + " не найден");
        return acc;
    }

    // ---------- Отчёты на Stream API ----------

    void printReports() {
        System.out.println("\n=== ОТЧЁТЫ (Stream API) ===");

        BigDecimal total = accounts.values().stream()
                .map(Account::getBalance)
                .reduce(BigDecimal.ZERO, BigDecimal::add);
        System.out.println("Суммарный капитал банка: " + total + " ₽");

        Optional<Account> richest = accounts.values().stream()
                .max(Comparator.comparing(Account::getBalance));
        richest.ifPresent(a -> System.out.println("Самый богатый клиент: " + a));

        Map<Boolean, List<Account>> partition = accounts.values().stream()
                .collect(Collectors.partitioningBy(
                        a -> a.getBalance().compareTo(new BigDecimal("10000")) >= 0));
        System.out.println("Счетов с балансом ≥ 10 000 ₽: " + partition.get(true).size());
        System.out.println("Счетов с балансом < 10 000 ₽: " + partition.get(false).size());

        System.out.println("Последние 5 транзакций:");
        history.stream()
                .skip(Math.max(0, history.size() - 5))
                .forEach(t -> System.out.printf("  %s → %s : %s ₽ (%s)%n",
                        t.fromId(), t.toId(), t.amount(), t.timestamp()));
    }

    // ---------- Консольное меню ----------

    private static final String MENU = """
            ┌────────────────────────────────┐
            │  БАНК «РОГА И КОПЫТА» — JDK 21 │
            ├────────────────────────────────┤
            │ 1. Открыть счёт                │
            │ 2. Пополнить счёт              │
            │ 3. Перевести средства          │
            │ 4. Показать все счета          │
            │ 5. Отчёты (Stream API)         │
            │ 0. Выход                       │
            └────────────────────────────────┘
            Ваш выбор: """;

    public static void main(String[] args) {
        BankApp bank = new BankApp();
        Scanner in = new Scanner(System.in);

        // Демо-данные, чтобы не вводить руками
        Account a1 = bank.openAccount("Иван Петров");
        Account a2 = bank.openAccount("Мария Сидорова");
        a1.deposit(new BigDecimal("50000"));
        a2.deposit(new BigDecimal("7500"));

        while (true) {
            System.out.print(MENU);
            String choice = in.nextLine().trim();
            try {
                switch (choice) {
                    case "1" -> {
                        System.out.print("Имя владельца: ");
                        bank.openAccount(in.nextLine().trim());
                    }
                    case "2" -> {
                        System.out.print("ID счёта и сумма через пробел: ");
                        String[] parts = in.nextLine().split("\\s+");
                        bank.require(Long.parseLong(parts[0]))
                            .deposit(new BigDecimal(parts[1]));
                    }
                    case "3" -> {
                        System.out.print("Откуда куда сумма (3 числа): ");
                        String[] p = in.nextLine().split("\\s+");
                        bank.transfer(Long.parseLong(p[0]),
                                      Long.parseLong(p[1]),
                                      new BigDecimal(p[2]));
                    }
                    case "4" -> bank.accounts.values().forEach(System.out::println);
                    case "5" -> bank.printReports();
                    case "0" -> {
                        System.out.println("До свидания!");
                        return;
                    }
                    default -> System.out.println("Нет такого пункта меню");
                }
            } catch (InsufficientFundsException | IllegalArgumentException
                     | NoSuchElementException | NumberFormatException
                     | ArrayIndexOutOfBoundsException e) {
                System.out.println("⛔ Ошибка: " + e.getMessage());
            }
        }
    }
}
