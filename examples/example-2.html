Пример 2: HTTP-сервер на JDK 21

JSON API без единой зависимости: HttpServer + виртуальные потоки

В JDK встроен HTTP-сервер (com.sun.net.httpserver, модуль jdk.httpserver) и современный HTTP-клиент (java.net.http, JDK 11+). С виртуальными потоками JDK 21 получается полноценный учебный REST API без Spring, без Maven, одним файлом.

Полный код (ApiServer.java)

java
import com.sun.net.httpserver.HttpExchange;
import com.sun.net.httpserver.HttpServer;

import java.io.IOException;
import java.io.OutputStream;
import java.net.InetSocketAddress;
import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.Executors;
import java.util.concurrent.atomic.AtomicLong;

/**
 * Учебный REST API «Справочник версий Java» на чистом JDK 21.
 * GET  /api/versions          — список версий (JSON)
 * GET  /api/versions/{id}     — одна версия
 * POST /api/versions          — добавить версию (JSON в теле)
 * GET  /api/stats             — статистика запросов
 * GET  /health                — проверка живости
 */
public class ApiServer {

    record JavaVersion(int id, String name, int year, boolean lts) {}

    // «База данных» в памяти — конкурентная, потокобезопасная
    private static final Map<Integer, JavaVersion> DB = new ConcurrentHashMap<>(Map.of(
            1, new JavaVersion(1, "Java 8",  2014, true),
            2, new JavaVersion(2, "Java 11", 2018, true),
            3, new JavaVersion(3, "Java 17", 2021, true),
            4, new JavaVersion(4, "Java 21", 2023, true),
            5, new JavaVersion(5, "Java 25", 2025, true)));
    private static final AtomicLong nextId = new AtomicLong(6);
    private static final AtomicLong requests = new AtomicLong();

    public static void main(String[] args) throws IOException {
        int port = args.length > 0 ? Integer.parseInt(args[0]) : 8080;

        HttpServer server = HttpServer.create(new InetSocketAddress(port), 0);

        // ★ JDK 21: виртуальный поток на каждый запрос — никаких пулов и настроек
        server.setExecutor(Executors.newVirtualThreadPerTaskExecutor());

        server.createContext("/api/versions", ApiServer::handleVersions);
        server.createContext("/api/stats", ex -> respond(ex, 200, """
                {"totalRequests": %d, "versionsInDb": %d, "time": "%s"}"""
                .formatted(requests.get(), DB.size(), LocalDateTime.now())));
        server.createContext("/health", ex -> respond(ex, 200, "{\"status\":\"UP\"}"));

        server.start();
        System.out.println("API запущен: http://localhost:" + port + "/api/versions");
    }

    /** Роутинг /api/versions с помощью switch по пути и методу (JDK 21 style). */
    static void handleVersions(HttpExchange ex) throws IOException {
        requests.incrementAndGet();
        String path = ex.getRequestURI().getPath();          // /api/versions[/id]
        String tail = path.length() > "/api/versions/".length() && path.startsWith("/api/versions/")
                ? path.substring("/api/versions/".length()) : "";

        try {
            switch (ex.getRequestMethod() + " " + (tail.isEmpty() ? "collection" : "item")) {
                case "GET collection" -> respond(ex, 200, listJson());
                case "GET item" -> {
                    int id = Integer.parseInt(tail);
                    JavaVersion v = DB.get(id);
                    if (v == null) respond(ex, 404, errorJson("Версия не найдена: " + id));
                    else respond(ex, 200, toJson(v));
                }
                case "POST collection" -> {
                    String body = new String(ex.getRequestBody().readAllBytes(), StandardCharsets.UTF_8);
                    JavaVersion v = parse(body);   // учебный парсер — в проде берите Jackson/Gson
                    DB.put(v.id(), v);
                    respond(ex, 201, toJson(v));
                }
                default -> respond(ex, 405, errorJson("Метод не поддерживается"));
            }
        } catch (NumberFormatException e) {
            respond(ex, 400, errorJson("Некорректный id"));
        }
    }

    // ---------- Минимум JSON-кода руками (без библиотек) ----------

    static String listJson() {
        StringBuilder sb = new StringBuilder("[");
        DB.values().stream()
          .sorted((a, b) -> Integer.compare(a.year(), b.year()))
          .forEach(v -> sb.append(toJson(v)).append(','));
        if (sb.length() > 1) sb.setLength(sb.length() - 1);
        return sb.append(']').toString();
    }

    static String toJson(JavaVersion v) {
        return """
                {"id":%d,"name":"%s","year":%d,"lts":%b}"""
                .formatted(v.id(), v.name(), v.year(), v.lts());
    }

    static String errorJson(String msg) {
        return "{\"error\":\"%s\"}".formatted(msg.replace("\"", "'"));
    }

    /** ОЧЕНЬ учебный парсер: {"name":"Java 22","year":2024,"lts":false} */
    static JavaVersion parse(String json) {
        String name = json.replaceAll(".*\"name\"\\s*:\\s*\"([^\"]+)\".*", "$1");
        int year = Integer.parseInt(json.replaceAll(".*\"year\"\\s*:\\s*(\\d+).*", "$1"));
        boolean lts = Boolean.parseBoolean(json.replaceAll(".*\"lts\"\\s*:\\s*(\\w+).*", "$1"));
        return new JavaVersion((int) nextId.getAndIncrement(), name, year, lts);
    }

    static void respond(HttpExchange ex, int status, String body) throws IOException {
        byte[] bytes = body.getBytes(StandardCharsets.UTF_8);
        ex.getResponseHeaders().set("Content-Type", "application/json; charset=utf-8");
        ex.sendResponseHeaders(status, bytes.length);
        try (OutputStream os = ex.getResponseBody()) {
            os.write(bytes);
        }
    }
}

Запуск и тестирование

bash
# Запуск (порт 8080):
$ java ApiServer.java
API запущен: http://localhost:8080/api/versions

# В другом терминале — проверяем curl'ом:
$ curl http://localhost:8080/api/versions
[{"id":1,"name":"Java 8","year":2014,"lts":true},...,{"id":5,"name":"Java 25","year":2025,"lts":true}]

$ curl http://localhost:8080/api/versions/4
{"id":4,"name":"Java 21","year":2023,"lts":true}

$ curl -X POST http://localhost:8080/api/versions \
       -d '{"name":"Java 26","year":2026,"lts":false}'
{"id":6,"name":"Java 26","year":2026,"lts":false}

$ curl http://localhost:8080/api/versions/999
{"error":"Версия не найдена: 999"}

Клиент на java.net.http (тестируем сервер кодом)

java
import java.net.URI;
import java.net.http.*;
import java.time.Duration;

HttpClient client = HttpClient.newBuilder()
        .version(HttpClient.Version.HTTP_2)          // HTTP/2 из коробки!
        .connectTimeout(Duration.ofSeconds(3))
        .build();

HttpRequest request = HttpRequest.newBuilder()
        .uri(URI.create("http://localhost:8080/api/versions/4"))
        .timeout(Duration.ofSeconds(5))
        .GET()
        .build();

// Синхронно:
HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
System.out.println(response.statusCode() + " " + response.body());

// Асинхронно (CompletableFuture):
client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
      .thenApply(HttpResponse::body)
      .thenAccept(System.out::println);

Нагрузочный мини-тест виртуальных потоков

java
// 10 000 параллельных клиентов — каждый ждёт свой ответ.
// С пулом платформенных потоков это сотни потоков ОС и секунды ожидания;
// с виртуальными — доли секунды и ~пики памяти в десятки МБ.
try (var pool = Executors.newVirtualThreadPerTaskExecutor()) {
    var futures = java.util.stream.IntStream.range(0, 10_000)
        .mapToObj(i -> pool.submit(() ->
            client.send(request, HttpResponse.BodyHandlers.ofString()).statusCode()))
        .toList();
    long ok = futures.stream().filter(f -> f.get() == 200).count();
    System.out.println("Успешных ответов: " + ok + "/10000");
}
📘 Примечание
Этот пример — идеальная основа для лабораторной работы 3.3 (динамический контент и CDN): добавьте в обработчик Thread.sleep(1000) («медленный Origin») и сравните TTFB напрямую и через Cloudflare.
⚠️ Внимание
com.sun.net.httpserver предназначен для разработки и тестов. В продакшене используйте полноценные серверы/фреймворки (Spring Boot, Quarkus, Helidon, Javalin) — но сам подход «виртуальный поток на запрос» в них уже стандарт (Spring Boot 3.2+, Tomcat 10.1+).
